   433	    private fun balancedJsonEnd(value: String, start: Int, open: Char, close: Char): Int? {
   434	        var depth = 0
   435	        var inString = false
   436	        var escaped = false
   437	        for (index in start until value.length) {
   438	            val char = value[index]
   439	            if (inString) {
   440	                when {
   441	                    escaped -> escaped = false
   442	                    char == '\\' -> escaped = true
   443	                    char == '"' -> inString = false
   444	                }
   445	                continue
   446	            }
   447	            when (char) {
   448	                '"' -> inString = true
   449	                open -> depth += 1
   450	                close -> {
   451	                    depth -= 1
   452	                    if (depth == 0) return index
   453	                }
   454	            }
   455	        }
   456	        return null
   457	    }
