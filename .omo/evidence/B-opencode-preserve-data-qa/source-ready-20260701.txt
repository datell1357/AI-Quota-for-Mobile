    12	class ProviderStateDebugReceiver : BroadcastReceiver() {
    13	    override fun onReceive(context: Context, intent: Intent?) {
    14	        if (intent?.action != ACTION_RESET_PROVIDER) return
    15	        val appContext = context.applicationContext
    16	        val requestedProvider = ProviderId.fromStorageId(
    17	            intent.getStringExtra(EXTRA_PROVIDER_ID)
    18	                ?: intent.getStringExtra(WidgetRefreshActions.EXTRA_PROVIDER_ID)
    19	        )
    20	        val providers = requestedProvider?.let(::listOf) ?: listOf(ProviderId.CLAUDE, ProviderId.CODEX, ProviderId.GEMINI, ProviderId.COPILOT)
    21	        providers.forEach { providerId ->
    22	            LocalUsageRepository(appContext).markSessionExpired(providerId, "Debug QA provider reset.")
    23	            if (providerId == ProviderId.GEMINI) {
    24	                DebugProviderSessionCookieStore.clear(appContext, providerId)
    25	            }
    26	        }
    27	        Log.i(TAG, "debugProviderReset=true providers=${providers.joinToString(",") { it.storageId }}")
    28	    }
    29	
    30	    companion object {
    31	        const val ACTION_RESET_PROVIDER = "com.aiquota.mobile.debug.RESET_PROVIDER"
    32	        const val EXTRA_PROVIDER_ID = "provider_id"
    33	        private const val TAG = "AIQuotaDebugState"
    34	    }
    35	}
